  (load "../common.scm")

  (define (%os name)
    (operating-system
      (bootloader <<bootloader>>)
      (file-systems (append <<file-systems>> %base-file-systems))
      (host-name name)
      (users (cons* <<user>> %base-user-accounts))
      (packages (append (pkgs+out <<package>>) %base-packages))
      (services (cons* <<service>> %base-services))))

  (compose <<transformation>> %os)

    (cuirass-remote-worker-configuration
     (server "ci.ultrarare.space:5555")
     (systems
      (cond
       ((target-x86? (or (%current-target-system)
                         (%current-system)))
        '("x86_64-linux" "i686-linux"))
       (else
        (list %current-system))))
     (publish-port 5558)
     (substitute-urls
      `(,@%default-substitute-urls
        "https://ci.ultrarare.space"))))
