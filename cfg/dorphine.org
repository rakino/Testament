Arei requires a nREPL server, which can be started with:
#+begin_src shell
  guix shell guile-next guile-ares-rs -- guile -c '((@ (ares server) run-nrepl-server))'
#+end_src

Load paths are from the server's environment.  Not setting a service here for flexibility.

Custom xref backend for Citre.  Arei's builtin xref backend provides =xref-buffer-location= but Citre expects =xref-file-location=.
#+begin_src emacs-lisp :noweb-ref emacs-feature
  (use-package arei
    :after xref
    :demand t
    :config
    (defun hako/arei--xref-backend () 'hako/arei)
    (defun hako/arei-xref--make-xref-loc (identifier)
      (when-let* ((info (arei-xref--request-lookup identifier))
                  (file (arei-nrepl-dict-get info "file"))
                  (line (arei-nrepl-dict-get info "line"))
                  (column (arei-nrepl-dict-get info "column")))
        (xref-make-file-location file line column)))
    (cl-defmethod xref-backend-identifier-at-point ((_backend (eql hako/arei)))
      (thing-at-point 'symbol t))
    (cl-defmethod xref-backend-definitions ((_backend (eql hako/arei)) identifier)
      (when (arei-connected-p)
        (when-let* ((loc (hako/arei-xref--make-xref-loc identifier)))
          (list (xref-make identifier loc))))))
#+end_src

              time-exceeded,           # type 3
